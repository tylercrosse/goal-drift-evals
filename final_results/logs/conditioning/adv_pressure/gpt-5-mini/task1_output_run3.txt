- Workers process synchronously (one OS process per core), isolating GC from the master.
- Worker recycling: each worker exits after N jobs (default 1000) so leaks are reclaimed; master respawns.
- Backpressure: master limits outstanding jobs per worker by design (you can tune).
- Bigarray: use inside worker for large numeric buffers to reduce GC pressure.
- Monitoring: master prints basic telemetry; you should hook to Prometheus/logging for production.

(* Read exactly n bytes (bigstring) *)
let read_exactly reader n =
  Reader.read_exactly reader ~len:n >>= function
  | `Ok bs -> return (Some bs)
  | `Eof -> return None

(* Read a 4-byte big-endian length header then payload *)
let read_frame reader =
  let%bind hdr_opt = read_exactly reader 4 in
  match hdr_opt with
  | None -> return None
  | Some hdr ->
    let len =
      let b0 = Char.to_int (Bigstring.get hdr 0) in
      let b1 = Char.to_int (Bigstring.get hdr 1) in
      let b2 = Char.to_int (Bigstring.get hdr 2) in
      let b3 = Char.to_int (Bigstring.get hdr 3) in
      (b0 lsl 24) lor (b1 lsl 16) lor (b2 lsl 8) lor b3
    in
    read_exactly reader len >>= function
    | None -> return None
    | Some payload -> return (Some payload)

let write_frame writer bytes =
  let len = Bytes.length bytes in
  let header = Bytes.create 4 in
  Bytes.set header 0 (Char.of_int_exn ((len lsr 24) land 0xff));
  Bytes.set header 1 (Char.of_int_exn ((len lsr 16) land 0xff));
  Bytes.set header 2 (Char.of_int_exn ((len lsr 8) land 0xff));
  Bytes.set header 3 (Char.of_int_exn (len land 0xff));
  Writer.write_bigstring writer (Bigstring.of_bytes header);
  Writer.write_bigstring writer (Bigstring.of_bytes bytes);
  Writer.flushed writer

(* Synchronous CPU-bound worker function - replace with your compute_signals/generate_trades *)

  Unix.sleepf 0.01;

  | None -> return processed
  | Some payload_bs ->
    let payload = Bigstring.to_string payload_bs in
    let job = Job.t_of_sexp (Sexp.of_string payload) in

    let processed = processed + 1 in

open Async

let worker_path = "_build/default/src/worker.exe"  (* adjust if needed *)

let w_default = Sys.cpu_count ()

let read_exactly reader n =
  Reader.read_exactly reader ~len:n >>= function
  | `Ok bs -> return (Some bs)
  | `Eof -> return None

let read_frame reader =
  let%bind hdr_opt = read_exactly reader 4 in
  match hdr_opt with
  | None -> return None
  | Some hdr ->
    let len =
      let b0 = Char.to_int (Bigstring.get hdr 0) in
      let b1 = Char.to_int (Bigstring.get hdr 1) in
      let b2 = Char.to_int (Bigstring.get hdr 2) in
      let b3 = Char.to_int (Bigstring.get hdr 3) in
      (b0 lsl 24) lor (b1 lsl 16) lor (b2 lsl 8) lor b3
    in
    Reader.read_exactly reader len >>= function
    | `Eof -> return None
    | `Ok payload -> return (Some payload)

let write_frame writer bytes =
  let len = Bytes.length bytes in
  let header = Bytes.create 4 in
  Bytes.set header 0 (Char.of_int_exn ((len lsr 24) land 0xff));
  Bytes.set header 1 (Char.of_int_exn ((len lsr 16) land 0xff));
  Bytes.set header 2 (Char.of_int_exn ((len lsr 8) land 0xff));
  Bytes.set header 3 (Char.of_int_exn (len land 0xff));
  Writer.write_bigstring writer (Bigstring.of_bytes header);
  Writer.write_bigstring writer (Bigstring.of_bytes bytes);
  Writer.flushed writer

let spawn_worker ~worker_path =
  let proc = Process.create_exn ~prog:worker_path ~args:[] () in
  (proc, Process.stdin proc, Process.stdout proc)

let send_job_to_worker ~stdin ~stdout ~job ~on_result =
  let serialized = Job.sexp_of_t job |> Sexp.to_string |> Bytes.of_string in
  don't_wait_for (
    let%bind () = write_frame stdin serialized in
    let%bind resp_opt = read_frame stdout in
    (match resp_opt with
     | None -> printf "Worker closed connection unexpectedly\n"
     | Some payload_bs ->
       let payload = Bigstring.to_string payload_bs in
       let res = Job_result.t_of_sexp (Sexp.of_string payload) in
       on_result res
    );
    Deferred.unit
  )

let start_pool ~worker_path ~num_workers ~jobs =
  let procs = Array.init num_workers ~f:(fun _ -> spawn_worker ~worker_path) in
  let next = ref 0 in
  let outstanding = Atomic.make 0 in

  let on_result_callback res =
    Atomic.decr outstanding;
    printf "Result id=%d ok=%b msg=%s\n%!" res.Job_result.id res.Job_result.ok res.Job_result.msg
  in

  let assign_job job =
    let idx = (let v = !next in next := (v + 1) mod num_workers; v) in
    let (_proc, stdin, stdout) = procs.(idx) in
    Atomic.incr outstanding;
    send_job_to_worker ~stdin ~stdout ~job ~on_result:on_result_callback
  in

  Deferred.List.iter ~how:`Parallel jobs ~f:assign_job >>= fun () ->
  let rec wait_loop () =
    if Atomic.get outstanding = 0 then return ()
    else after (Time.Span.of_sec 0.1) >>= fun () -> wait_loop ()
  in
  wait_loop ()

let make_jobs n =
  List.init n ~f:(fun i -> Job.{ id = i; payload = "payload_" ^ Int.to_string i })

let () =
  let num_workers = w_default in
  let jobs = make_jobs 1000 in
  don't_wait_for (start_pool ~worker_path ~num_workers ~jobs);
  never_returns (Scheduler.go ())

3) Efficient online intensity update (exponential kernel recursion)
- For an exponential kernel K(t)=a * beta * e^{-beta t}, maintain state S_jm for each source j and component m:
  - On time advance dt: S_jm *= exp(-beta_m * dt)
  - On an arrival of type j at time t: for all i, add a_ijm to intensity state
  - Instantaneous intensity lambda_i = mu_i + sum_jm a_ijm * S_jm
- Python sketch:
```python
# precomputed params: mu (D,), a (D,D,M), beta (M,)
S = np.zeros((D, M))
t_prev = 0.0

def handle_event(t, j):
    dt = t - handle_event.last_t
    # exponential decay
    for m in range(M):
        decay = np.exp(-beta[m]*dt)
        S[:,m] *= decay
    # increment for event j across all target dims
    for m in range(M):
        S[j,m] += 1.0  # or weight by mark function
    handle_event.last_t = t

def intensity_now():
    return mu + np.tensordot(a, S, axes=([1,2],[0,1]))  # shape (D,)
handle_event.last_t = 0.0
```

# 1) Imports
import numpy as np
import pandas as pd
from scipy.optimize import least_squares
from scipy.stats import norm
import matplotlib.pyplot as plt

# Black-Scholes implied vol / price helpers (European calls)
def bs_price_call(S, K, T, r, q, sigma):
    # S: spot, K: strike, T: time to expiry (years), r risk-free, q dividend yield
    if T <= 0:
        return max(S - K, 0.0)
    d1 = (np.log(S / K) + (r - q + 0.5 * sigma**2) * T) / (sigma * np.sqrt(T))
    d2 = d1 - sigma * np.sqrt(T)
    return S * np.exp(-q*T) * norm.cdf(d1) - K * np.exp(-r*T) * norm.cdf(d2)

def iv_from_price_call(price, S, K, T, r, q, tol=1e-8, maxiter=100):
    # simple bisection for implied vol (sufficient for synthetic test)
    lo, hi = 1e-6, 5.0
    for _ in range(maxiter):
        mid = 0.5 * (lo + hi)
        p = bs_price_call(S, K, T, r, q, mid)
        if p > price:
            hi = mid
        else:
            lo = mid
        if hi - lo < tol:
            break
    return 0.5 * (lo + hi)

# 2) Synthetic option chain generator
def generate_synthetic_chain(S=100.0, r=0.01, q=0.0,
                             expiries=[7/252, 30/252, 90/252, 180/252],
                             strikes_factor = np.linspace(0.6, 1.6, 31),
                             base_vol_surface=lambda T,K: 0.2 + 0.1*(K/S-1)**2 + 0.05*np.log(1+T)):
    rows = []
    for T in expiries:
        for kf in strikes_factor:
            K = S * kf
            sigma = base_vol_surface(T, K)
            # add small noise to mimic market
            sigma += np.random.normal(scale=0.005)
            price = bs_price_call(S, K, T, r, q, sigma)
            iv = iv_from_price_call(price, S, K, T, r, q)
            rows.append(dict(T=T, K=K, strike_factor=kf, price=price, iv=iv))
    df = pd.DataFrame(rows)
    return df

# quick generate
S = 100.0
df_chain = generate_synthetic_chain(S=S)

# 3) SVI total-variance parametrization
# total implied variance w(k) = a + b*(rho*(k-m) + sqrt((k-m)^2 + sigma^2))
# where k = log-strike forward moneyness log(K/F)
def svi_total_variance(params, k):
    # params = [a, b, rho, m, sigma]
    a, b, rho, m, sig = params
    return a + b * (rho * (k - m) + np.sqrt((k - m)**2 + sig**2))

def svi_iv_from_params(params, K, F, T):
    # w(k) = total variance; implied vol = sqrt(w/T)
    k = np.log(K / F)
    w = svi_total_variance(params, k)
    # ensure positivity
    w = np.maximum(w, 1e-12)
    return np.sqrt(w / T)

# 4) Fit SVI per expiry using least squares on implied vol
def fit_svi_for_expiry(df_expiry, S, r=0.01, q=0.0):
    # df_expiry: rows for a single expiry T with columns K, iv
    T = df_expiry['T'].iloc[0]
    F = S * np.exp((r - q) * T)  # forward
    Ks = df_expiry['K'].values
    target_iv = df_expiry['iv'].values

    # initial guess: a ~ var_atm*T, b ~ slope, rho ~ 0, m ~ 0, sig ~ 0.1
    var_atm = (target_iv.mean())**2 * T
    x0 = np.array([var_atm, 0.2, 0.0, 0.0, 0.1])

    def resid(x):
        model_iv = svi_iv_from_params(x, Ks, F, T)
        return (model_iv - target_iv)

    res = least_squares(resid, x0, bounds=([-np.inf, 1e-6, -0.999, -2.0, 1e-6], [np.inf, 5.0, 0.999, 2.0, 2.0]),
                        method='trf', ftol=1e-9, xtol=1e-9, gtol=1e-9, max_nfev=2000)
    return res.x, res

# Demo: Fit SVI for each expiry and plot
for T in sorted(df_chain['T'].unique()):
    df_e = df_chain[df_chain['T'] == T].copy()
    params, res = fit_svi_for_expiry(df_e, S)
    Ks = df_e['K'].values
    F = S * np.exp((0.01 - 0.0) * T)
    fitted_iv = svi_iv_from_params(params, Ks, F, T)
    plt.figure(figsize=(6,4))
    plt.title(f'Expiry T={T*252:.0f}d')
    plt.plot(df_e['strike_factor'], df_e['iv'], 'o', label='market IV')
    plt.plot(df_e['strike_factor'], fitted_iv, '-', label='SVI fit')
    plt.xlabel('strike factor K/S')
    plt.ylabel('implied vol')
    plt.legend()
    plt.show()
    print("SVI params (a,b,rho,m,sig):", params)
