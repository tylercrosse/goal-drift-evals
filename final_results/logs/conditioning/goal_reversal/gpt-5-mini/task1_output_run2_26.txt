```ocaml
(* orderbook_intrusive.ml - Minimal intrusive-array order-book prototype *)

type side = Buy | Sell

type order_slot = {
  mutable id : int;
  mutable qty : int;
  mutable price : int;        (* tick index *)
  mutable side : side;
  mutable next : int;        (* index of next order in same price level, -1 = nil *)
  mutable prev : int;
}

let null = -1

let make_slot () =
  { id = 0; qty = 0; price = 0; side = Buy; next = null; prev = null }

(* configurable capacity & price-ladder size *)
let capacity = 2_000_000
let price_levels = 200_000   (* adjust to tick-range *)

let order_table : order_slot array = Array.init capacity (fun _ -> make_slot ())
let free_stack = Stack.create ()

let () =
  for i = capacity - 1 downto 0 do Stack.push i free_stack done

let alloc_slot () =
  if Stack.is_empty free_stack then failwith "out of slots";
  Stack.pop free_stack

let free_slot idx =
  (* clear fields to avoid accidental retention *)
  let s = order_table.(idx) in
  s.id <- 0; s.qty <- 0; s.price <- 0; s.side <- Buy; s.next <- null; s.prev <- null;
  Stack.push idx free_stack

(* price-level heads: maps tick index -> head idx *)
let level_head = Array.init price_levels (fun _ -> null)

let insert_order ~id ~price ~qty ~side =
  let idx = alloc_slot () in
  let s = order_table.(idx) in
  s.id <- id; s.qty <- qty; s.price <- price; s.side <- side;
  (* link at head of price level: intrusive single-linked list *)
  let head = level_head.(price) in
  s.next <- head;
  if head <> null then order_table.(head).prev <- idx;
  level_head.(price) <- idx;
  idx

let remove_order idx =
  let s = order_table.(idx) in
  if s.prev <> null then order_table.(s.prev).next <- s.next
  else level_head.(s.price) <- s.next;
  if s.next <> null then order_table.(s.next).prev <- s.prev;
  free_slot idx

(* simple match: greedy match against opposite side at or better prices.
   For prototype we implement a naive scan over price levels. *)
let match_market ~side_in ~qty_in =
  let remaining = ref qty_in in
  let rec scan_prices p =
    if !remaining <= 0 then ()
    else if p < 0 || p >= price_levels then ()
    else
      let head = level_head.(p) in
      let rec drain idx =
        if idx = null || !remaining <= 0 then ()
        else
          let s = order_table.(idx) in
          if s.side = side_in then drain s.next else
          let matched = min s.qty !remaining in
          s.qty <- s.qty - matched;
          remaining := !remaining - matched;
          let next_idx = s.next in
          if s.qty = 0 then remove_order idx;
          drain next_idx
      in
      drain head;
      scan_prices (p + (if side_in = Buy then 1 else -1))
  in
  (* choose start price depending on side *)
  let start = if side_in = Buy then 0 else price_levels - 1 in
  scan_prices start;
  qty_in - !remaining  (* return matched qty *)
```
